# 2. 三组之间的死亡率比较（重点！）
# ============================================================

# 2.1 计算各组的死亡率和生存率
death_table <- table(data$PLT_RR_tertile, data$Death)
print(death_table)

# 计算各组的死亡率
death_prop <- prop.table(death_table, margin = 1) * 100
print(death_prop)

# 创建死亡率的汇总表
death_summary <- data.frame(
  Group = rownames(death_table),
  Alive = death_table[, "Alive"],
  Dead = death_table[, "Dead"],
  Total = rowSums(death_table),
  Mortality_Rate = round(death_prop[, "Dead"], 2)
)
print(death_summary)

# 2.2 卡方检验比较三组死亡率
chisq_test_death <- chisq.test(death_table)
print(chisq_test_death)

# 如果卡方检验显著，进行两两比较
if (chisq_test_death$p.value < 0.05) {
  cat("\n卡方检验显著，进行两两比较...\n")
  
  # 两两比较函数
  pairwise_death_comparison <- function(data, group_var, outcome_var) {
    groups <- levels(data[[group_var]])
    results <- data.frame(
      Group1 = character(),
      Group2 = character(),
      P_value = numeric(),
      stringsAsFactors = FALSE
    )
    
    for (i in 1:(length(groups)-1)) {
      for (j in (i+1):length(groups)) {
        subset_data <- data[data[[group_var]] %in% c(groups[i], groups[j]), ]
        tbl <- table(subset_data[[group_var]], subset_data[[outcome_var]])
        test <- chisq.test(tbl)
        results <- rbind(results, data.frame(
          Group1 = groups[i],
          Group2 = groups[j],
          P_value = test$p.value
        ))
      }
    }
    return(results)
  }
  
  pairwise_death <- pairwise_death_comparison(data, "PLT_RR_tertile", "Death")
  # 应用Bonferroni校正
  pairwise_death$P_adjusted <- p.adjust(pairwise_death$P_value, method = "bonferroni")
  pairwise_death$Significant <- pairwise_death$P_adjusted < 0.05
  print(pairwise_death)
}

# 4. 详细的统计分析
# ============================================================

# 创建函数来对每个变量进行三组比较
compare_by_tertile <- function(data, var, group_var) {
  # 检查是否为数值变量
  if (is.numeric(data[[var]])) {
    # 使用Kruskal-Wallis检验
    tryCatch({
      test <- kruskal.test(as.formula(paste(var, "~", group_var)), data = data)
      p_val <- test$p.value
      method <- "Kruskal-Wallis"
      return(data.frame(Variable = var, P_value = p_val, Method = method, Type = "Numeric"))
    }, error = function(e) {
      return(data.frame(Variable = var, P_value = NA, Method = "Error", Type = "Numeric"))
    })
  } else {
    # 分类变量使用卡方检验
    tryCatch({
      tbl <- table(data[[var]], data[[group_var]])
      if (any(chisq.test(tbl)$expected < 5)) {
        test <- fisher.test(tbl, simulate.p.value = TRUE)
        p_val <- test$p.value
        method <- "Fisher's Exact"
      } else {
        test <- chisq.test(tbl)
        p_val <- test$p.value
        method <- "Chi-square"
      }
      return(data.frame(Variable = var, P_value = p_val, Method = method, Type = "Categorical"))
    }, error = function(e) {
      return(data.frame(Variable = var, P_value = NA, Method = "Error", Type = "Categorical"))
    })
  }
}

# 对每个变量进行检验（包括Death）
results_list <- list()
for (var in vars_to_compare) {
  if (var %in% colnames(data) && var != "PLT_RR_tertile") {
    result <- compare_by_tertile(data, var, "PLT_RR_tertile")
    results_list[[var]] <- result
  }
}

# 合并结果
results_df <- do.call(rbind, results_list)

# 添加FDR校正
results_df$P_adjusted <- p.adjust(results_df$P_value, method = "fdr")
results_df$Significant <- results_df$P_adjusted < 0.05

# 按P值排序
results_df <- results_df[order(results_df$P_value), ]

# 突出显示Death的结果
print("Death变量的统计检验结果：")
print(results_df[results_df$Variable == "Death", ])

print("\n所有变量的统计检验结果（按P值排序）：")
print(results_df)

# 导出详细的检验结果
write.csv(results_df, "PLT_RR_tertile_statistical_tests.csv", row.names = FALSE)
cat("详细统计检验结果已保存为 PLT_RR_tertile_statistical_tests.csv\n")
