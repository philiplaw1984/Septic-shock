# Safely extract nonlinearity test P-value
extract_p_value <- function(anova_result, term_name = "PLT_RR") {
  if (term_name %in% rownames(anova_result)) {
    return(anova_result[term_name, "P"])
  } else {
    # Try fuzzy matching
    row_names <- rownames(anova_result)
    matching_rows <- grep(term_name, row_names, value = TRUE)
    if (length(matching_rows) > 0) {
      return(anova_result[matching_rows[1], "P"])
    } else {
      return(NA)
    }
  }
}

# 5.3 Likelihood ratio test (manual calculation)
cat("\n[Likelihood Ratio Test: Linear vs RCS (3 knots)]\n")

# Build linear model for comparison (using lrm)
linear_model_rms <- lrm(Death_binary ~ PLT_RR + Age + Sex + hypertension + 
                          CHD + DM + Smoking + BUN + Scr + Albumin + Hemoglobin,
                        data = data, x = TRUE, y = TRUE)

# Extract log-likelihoods
ll_linear <- logLik(linear_model_rms)
ll_rcs <- logLik(rcs_model_adj)

if (is.numeric(ll_linear) && is.numeric(ll_rcs)) {
  # Calculate likelihood ratio statistic
  lr_stat <- -2 * (as.numeric(ll_linear) - as.numeric(ll_rcs))
  
  # Calculate degrees of freedom difference
  df_linear <- length(coef(linear_model_rms))
  df_rcs <- length(coef(rcs_model_adj))
  df_diff <- df_rcs - df_linear
  
  if (df_diff > 0) {
    # Calculate P-value
    lrt_p_value <- 1 - pchisq(lr_stat, df_diff)
    
    cat(sprintf("Linear model log-likelihood: %.4f\n", as.numeric(ll_linear)))
    cat(sprintf("RCS model log-likelihood: %.4f\n", as.numeric(ll_rcs)))
    cat(sprintf("Likelihood ratio statistic (LR χ²): %.4f\n", lr_stat))
    cat(sprintf("Degrees of freedom: %d\n", df_diff))
    cat(sprintf("P-value: %.4f\n", lrt_p_value))
  } else {
    cat("Warning: Model df difference is negative, models may not be properly nested\n")
    lrt_p_value <- NA
  }
} else {
  cat("Warning: Unable to extract log-likelihood values\n")
  lrt_p_value <- NA
}

# 5.4 Wald test (joint test of RCS nonlinear terms)
cat("\n[Wald Test: Joint Test of RCS Nonlinear Terms]\n")

# Get coefficients and variance-covariance matrix
coef_rcs <- coef(rcs_model_adj)
vcov_rcs <- vcov(rcs_model_adj)

# Find RCS-related nonlinear terms (exclude intercept and linear term)
rcs_names <- grep("PLT_RR", names(coef_rcs), value = TRUE)
# If there are multiple PLT_RR terms, take all except the first as nonlinear terms
if (length(rcs_names) > 1) {
  nonlin_indices <- which(names(coef_rcs) %in% rcs_names[-1])
} else {
  nonlin_indices <- which(names(coef_rcs) %in% rcs_names)
}

if (length(nonlin_indices) > 1) {
  # Wald test
  beta_nonlin <- coef_rcs[nonlin_indices]
  vcov_nonlin <- vcov_rcs[nonlin_indices, nonlin_indices]
  
  # Calculate Wald statistic
  wald_stat <- t(beta_nonlin) %*% solve(vcov_nonlin) %*% beta_nonlin
  wald_p <- 1 - pchisq(wald_stat, length(nonlin_indices))
  
  cat(sprintf("Wald test statistic: %.4f\n", wald_stat))
  cat(sprintf("Degrees of freedom: %d\n", length(nonlin_indices)))
  cat(sprintf("P-value: %.4f\n", wald_p))
} else {
  cat("Insufficient nonlinear terms for joint test\n")
  wald_p <- NA
}

# Function to safely predict with rms models
safe_predict <- function(model, newdata, type = "fitted") {
  tryCatch({
    # Use predict.lrm directly
    pred <- predict(model, newdata = newdata, type = type)
    return(pred)
  }, error = function(e) {
    cat("Error in prediction:", e$message, "\n")
    return(NULL)
  })
}

find_optimal_cutpoint <- function(data, outcome, predictor, adjust_vars = NULL, 
                                   n_points = 100) {
  segmented_logit <- function(cutpoint) {
    data$predictor_below <- ifelse(data[[predictor]] <= cutpoint, data[[predictor]], cutpoint)
    data$predictor_above <- ifelse(data[[predictor]] > cutpoint, data[[predictor]] - cutpoint, 0)
    
    if (!is.null(adjust_vars)) {
      formula <- as.formula(paste(outcome, "~ predictor_below + predictor_above +", 
                                   paste(adjust_vars, collapse = " + ")))
    } else {
      formula <- as.formula(paste(outcome, "~ predictor_below + predictor_above"))
    }
    
    model <- glm(formula, data = data, family = binomial)
    return(-logLik(model))
  }
  
  plt_rr_range <- quantile(data[[predictor]], probs = c(0.1, 0.9), na.rm = TRUE)
  cutpoints <- seq(plt_rr_range[1], plt_rr_range[2], length.out = n_points)
  
  loglik_values <- sapply(cutpoints, function(cp) {
    tryCatch(segmented_logit(cp), error = function(e) Inf)
  })
  
  if (any(is.infinite(loglik_values))) {
    loglik_values[is.infinite(loglik_values)] <- max(loglik_values[is.finite(loglik_values)], na.rm = TRUE)
  }
  
  best_idx <- which.min(loglik_values)
  best_cutpoint <- cutpoints[best_idx]
  
  return(list(
    best_cutpoint = best_cutpoint,
    cutpoints = cutpoints,
    loglik_values = loglik_values
  ))
}

adjust_vars <- c("Age", "Sex", "hypertension", "CHD", "DM", "Smoking", 
                 "BUN", "Scr", "Albumin", "Hemoglobin")
