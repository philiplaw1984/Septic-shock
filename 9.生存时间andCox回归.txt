# ================================================================
# 完整分析流程：单因素COX筛选 + 多因素COX + 生存曲线
# 自动读取所有变量（无需手动输入）
# ================================================================

# ================================================================
# Part 0: 加载必要的包
# ================================================================

packages <- c("readxl", "survival", "survminer", "dplyr", "ggplot2", "car")
for (pkg in packages) {
  if (!require(pkg, character.only = TRUE)) {
    install.packages(pkg)
    library(pkg, character.only = TRUE)
  }
}

library(readxl)
library(survival)
library(survminer)
library(dplyr)
library(ggplot2)
library(car)  # 用于VIF检验

cat("========== 所有包加载成功！ ==========\n\n")


# ================================================================
# Part 1: 读取数据
# ================================================================

df <- read_excel("DD3.xlsx", sheet = "Sheet1")

cat("========== 数据读取成功 ==========\n")
cat("原始数据维度:", dim(df), "\n")
cat("所有列名:\n")
print(colnames(df))
cat("\n")


# ================================================================
# Part 2: 数据预处理（90天截断 + 剔除缺失值）
# ================================================================

df_analysis <- df %>%
  mutate(
    Days_90 = ifelse(Days > 90, 90, Days),
    Death_90 = ifelse(Days > 90, 0, Death)
  ) %>%
  filter(
    !is.na(PLT_RR), 
    !is.na(Days_90), 
    !is.na(Death_90)
  )

cat("========== 数据预处理完成 ==========\n")
cat("最终分析样本量:", nrow(df_analysis), "\n")
cat("90天内死亡人数:", sum(df_analysis$Death_90), "\n")
cat("90天内删失人数:", sum(df_analysis$Death_90 == 0), "\n")
cat("PLT_RR范围:", range(df_analysis$PLT_RR), "\n")
cat("PLT_RR中位数:", median(df_analysis$PLT_RR), "\n\n")


# ================================================================
# Part 3: 自动识别所有变量（修正版：正确排除结局变量）
# ================================================================

# ---------- 3.1 定义需要排除的变量 ----------
# 【关键修正】必须排除原始结局变量和新生成的结局变量
exclude_vars <- c(
  "Days",        # 原始生存时间
  "Death",       # 原始死亡状态
  "Days_90",     # 90天截断生存时间（结局变量！）
  "Death_90"     # 90天死亡状态（结局变量！）
  # 根据你的数据继续添加需要排除的列名，如 "ID", "PatientID" 等
)

# ---------- 3.2 自动提取所有候选变量 ----------
# 选择所有数值型变量，并排除掉 exclude_vars
all_vars <- df_analysis %>%
  select(where(is.numeric)) %>%
  select(-any_of(exclude_vars)) %>%
  colnames()

# 如果 PLT_RR、PLT、RR 不在其中（因为不是数值型），手动加入
force_vars <- c("PLT_RR", "PLT", "RR", "Age")

# 确保强制变量在 all_vars 中
for (var in force_vars) {
  if (!var %in% all_vars && var %in% colnames(df_analysis)) {
    all_vars <- c(all_vars, var)
  }
}

# 候选变量 = 全部变量 - 强制保留的变量
candidate_vars <- setdiff(all_vars, force_vars)

cat("========== 变量识别完成 ==========\n")
cat("全部候选变量数:", length(all_vars), "\n")
cat("强制保留变量数:", length(force_vars), "\n")
cat("待筛选变量数:", length(candidate_vars), "\n")
cat("\n强制保留变量:", paste(force_vars, collapse = ", "), "\n")
cat("\n待筛选变量（前20个）:\n")
print(head(candidate_vars, 20))
if (length(candidate_vars) > 20) {
  cat("... 还有", length(candidate_vars) - 20, "个变量\n")
}
cat("\n")


# ================================================================
# Part 4: 单因素COX回归筛选
# ================================================================

cat("========== 开始单因素COX回归筛选 ==========\n")

# ---------- 4.1 对每个候选变量进行单因素COX ----------
univ_results <- data.frame(
  Variable = character(),
  N = integer(),
  HR = numeric(),
  HR_lower = numeric(),
  HR_upper = numeric(),
  P_value = numeric(),
  stringsAsFactors = FALSE
)

# 如果有候选变量才执行
if (length(candidate_vars) > 0) {
  
  for (var in candidate_vars) {
    # 跳过非数值变量或全是缺失的变量
    if (!var %in% colnames(df_analysis)) next
    if (sum(!is.na(df_analysis[[var]])) < 10) next  # 至少10个有效值
    
    tryCatch({
      # 构建单因素公式
      formula_univ <- as.formula(paste("Surv(Days_90, Death_90) ~", var))
      model_univ <- coxph(formula_univ, data = df_analysis)
      
      # 提取结果
      coef_summary <- summary(model_univ)
      hr <- exp(coef(model_univ))
      ci <- exp(confint(model_univ))
      p_val <- coef_summary$coefficients[, "Pr(>|z|)"]
      n_obs <- model_univ$n
      
      # 存入结果表
      univ_results <- rbind(univ_results, data.frame(
        Variable = var,
        N = n_obs,
        HR = round(hr, 3),
        HR_lower = round(ci[, 1], 3),
        HR_upper = round(ci[, 2], 3),
        P_value = round(p_val, 4)
      ))
    }, error = function(e) {
      # 如果模型拟合失败，跳过该变量
      cat("  跳过变量", var, "(拟合失败)\n")
    })
  }
  
  # ---------- 4.2 按P值排序 ----------
  univ_results <- univ_results[order(univ_results$P_value), ]
  
  # 添加显著性标记
  univ_results$Significance <- ifelse(univ_results$P_value < 0.001, "***",
                               ifelse(univ_results$P_value < 0.01, "**",
                               ifelse(univ_results$P_value < 0.05, "*", "")))
  
  cat("\n========== 单因素COX回归结果 ==========\n")
  print(univ_results)
  cat("\n")
  
  # 保存完整单因素结果
  write.csv(univ_results, "01_Univariate_COX_results.csv", row.names = FALSE)
  cat("单因素COX结果已保存: 01_Univariate_COX_results.csv\n\n")
  
  # ---------- 4.3 筛选P < 阈值 的变量进入多因素 ----------
  # 根据事件数自动调整阈值
  event_count <- sum(df_analysis$Death_90)
  
  if (event_count >= 100) {
    p_threshold <- 0.05
  } else if (event_count >= 50) {
    p_threshold <- 0.1
  } else if (event_count >= 20) {
    p_threshold <- 0.2
  } else {
    p_threshold <- 0.25
  }
  
  cat("========== 单因素筛选结果 ==========\n")
  cat("事件数:", event_count, "\n")
  cat("自动选择的P值阈值:", p_threshold, "\n")
  
  significant_vars <- univ_results$Variable[univ_results$P_value < p_threshold]
  
  cat("显著变量数:", length(significant_vars), "\n")
  if (length(significant_vars) > 0) {
    cat("显著变量:\n")
    print(significant_vars)
  } else {
    cat("警告：没有变量达到P <", p_threshold, "的显著性水平\n")
    cat("建议：手动调整阈值或检查数据\n")
    # 如果P<阈值没变量，自动尝试P<0.25
    p_threshold <- 0.25
    significant_vars <- univ_results$Variable[univ_results$P_value < p_threshold]
    cat("\n改用P < 0.25，显著变量数:", length(significant_vars), "\n")
    if (length(significant_vars) > 0) {
      print(significant_vars)
    }
  }
  cat("\n")
  
} else {
  cat("没有候选变量需要筛选\n")
  significant_vars <- c()
}


# ================================================================
# Part 5: 准备进入多因素COX的变量列表
# ================================================================

# ---------- 5.1 合并变量：单因素显著 + 强制保留 ----------
final_vars <- unique(c(significant_vars, force_vars))

# 只保留数据中实际存在的变量
final_vars <- final_vars[final_vars %in% colnames(df_analysis)]

# 【安全检查】确保结局变量没有被误纳入
final_vars <- setdiff(final_vars, c("Days", "Death", "Days_90", "Death_90"))

cat("========== 最终COX模型变量 ==========\n")
cat("最终纳入变量数:", length(final_vars), "\n")
cat("最终变量列表:\n")
print(final_vars)

# 检查变量数是否过多（事件数/10）
event_count <- sum(df_analysis$Death_90)
max_vars_allowed <- floor(event_count / 10)
if (length(final_vars) > max_vars_allowed && max_vars_allowed > 0) {
  cat("\n警告：变量数 (", length(final_vars), ") 超过建议值 (", max_vars_allowed, ")\n")
  cat("建议：增加P值阈值（如从", p_threshold, "降到0.05）或考虑逐步回归\n")
}
cat("\n")


# ================================================================
# Part 6: 构建最终多因素COX模型
# ================================================================

# ---------- 6.1 准备最终数据 ----------
df_final <- df_analysis[, final_vars]

# 处理缺失值（用中位数填充）
df_final <- df_final %>%
  mutate(across(where(is.numeric), ~ ifelse(is.na(.), median(., na.rm = TRUE), .)))

# 添加生存时间和事件（注意：这两个变量不参与建模，只在公式左侧）
df_final$Days_90 <- df_analysis$Days_90
df_final$Death_90 <- df_analysis$Death_90

# ---------- 6.2 拟合多因素COX ----------
formula_str <- paste("Surv(Days_90, Death_90) ~", paste(final_vars, collapse = " + "))
cat("========== 多因素COX模型 ==========\n")
cat("模型公式:\n", formula_str, "\n\n")

final_cox <- coxph(as.formula(formula_str), data = df_final)

cat("========== 多因素COX回归结果 ==========\n")
print(summary(final_cox))
cat("\n")

# ---------- 6.3 提取HR汇总表 ----------
hr_table <- data.frame(
  Variable = final_vars,
  HR = round(exp(coef(final_cox)), 3),
  HR_lower = round(exp(confint(final_cox))[, 1], 3),
  HR_upper = round(exp(confint(final_cox))[, 2], 3),
  P_value = round(summary(final_cox)$coefficients[, "Pr(>|z|)"], 4)
)

# 添加显著性标记
hr_table$Significance <- ifelse(hr_table$P_value < 0.001, "***",
                          ifelse(hr_table$P_value < 0.01, "**",
                          ifelse(hr_table$P_value < 0.05, "*", "")))

cat("========== HR汇总表 ==========\n")
print(hr_table)
cat("\n")

# 保存HR表
write.csv(hr_table, "02_Final_COX_HR_table.csv", row.names = FALSE)
cat("HR汇总表已保存: 02_Final_COX_HR_table.csv\n\n")

# ---------- 6.4 VIF检验（检查共线性） ----------
if (length(final_vars) >= 2) {
  cat("========== VIF共线性检验 ==========\n")
  tryCatch({
    vif_values <- vif(final_cox)
    print(vif_values)
    cat("\nVIF > 5 表示存在中度共线性，VIF > 10 表示严重共线性\n")
  }, error = function(e) {
    cat("VIF检验失败（可能由于分类变量或模型复杂度）\n")
  })
  cat("\n")
}


# ================================================================
# Part 7: PLT_RR三分位生存曲线（未调整）
# ================================================================

# ---------- 7.1 三分位分组 ----------
tertiles <- quantile(df_analysis$PLT_RR, probs = c(0, 1/3, 2/3, 1), na.rm = TRUE)

df_analysis$PLT_group <- cut(
  df_analysis$PLT_RR, 
  breaks = tertiles, 
  include.lowest = TRUE,
  labels = c("Low (T1)", "Medium (T2)", "High (T3)")
)

df_analysis$PLT_group <- relevel(df_analysis$PLT_group, ref = "Low (T1)")

cat("========== PLT_RR三分位分组 ==========\n")
cat("三分位界值:\n")
print(tertiles)
cat("\n各组样本量:\n")
print(table(df_analysis$PLT_group))
cat("\n")

# ---------- 7.2 三分位COX（未调整，用于HR标注） ----------
cox_tertile <- coxph(Surv(Days_90, Death_90) ~ PLT_group, data = df_analysis)
hr_tert <- exp(coef(cox_tertile))
ci_tert <- exp(confint(cox_tertile))
p_tert <- summary(cox_tertile)$coefficients[, "Pr(>|z|)"]

hr_annotation <- sprintf(
  "PLT_RR (Tertile Split)\nMedium vs Low: HR = %.2f (95%% CI: %.2f-%.2f), p = %.3f\nHigh vs Low: HR = %.2f (95%% CI: %.2f-%.2f), p = %.3f",
  hr_tert[1], ci_tert[1, 1], ci_tert[1, 2], p_tert[1],
  hr_tert[2], ci_tert[2, 1], ci_tert[2, 2], p_tert[2]
)

# ---------- 7.3 绘制KM曲线（未调整） ----------
fit_km <- survfit(Surv(Days_90, Death_90) ~ PLT_group, data = df_analysis)
logrank_p <- surv_pvalue(fit_km)$pval

p_km <- ggsurvplot(
  fit_km,
  data = df_analysis,
  pval = TRUE,
  pval.method = TRUE,
  conf.int = TRUE,
  conf.int.style = "step",
  risk.table = TRUE,
  risk.table.col = "strata",
  risk.table.height = 0.30,
  surv.median.line = "hv",
  legend.labs = c("Low (T1)", "Medium (T2)", "High (T3)"),
  legend.title = "PLT_RR Group",
  xlab = "Survival Time (Days)",
  ylab = "Survival Probability",
  title = "Kaplan-Meier Survival Curve by PLT_RR (Tertile Split)",
  subtitle = paste0("Log-rank p = ", round(logrank_p, 4)),
  palette = c("#2E9FDF", "#F39C12", "#E74C3C"),
  xlim = c(0, 90),
  break.time.by = 10,
  ggtheme = theme_bw() +
    theme(
      legend.position = c(0.82, 0.82),
      legend.background = element_rect(fill = "white", color = "black", size = 0.3),
      legend.title = element_text(size = 10),
      legend.text = element_text(size = 9),
      plot.title = element_text(hjust = 0.5, size = 14, face = "bold"),
      plot.subtitle = element_text(hjust = 0.5, size = 10),
      axis.title = element_text(size = 12),
      axis.text = element_text(size = 10)
    )
)

p_km$plot <- p_km$plot + 
  annotate(
    "text",
    x = 2,
    y = 0.08,
    label = hr_annotation,
    hjust = 0,
    vjust = 0,
    size = 3.2,
    fontface = "plain",
    color = "black"
  )

print(p_km)

png("03_KM_curve_PLT_RR_unadjusted.png", width = 3000, height = 3600, res = 400)
print(p_km)
dev.off()
cat("未调整生存曲线已保存: 03_KM_curve_PLT_RR_unadjusted.png\n\n")


# ================================================================
# Part 8: 调整后的生存曲线（Adjusted Survival Curve）
# ================================================================

# ---------- 8.1 检查是否需要绘制 ----------
if (length(final_vars) > 0 && "PLT_RR" %in% final_vars) {
  
  cat("\n========== 开始绘制调整后的生存曲线 ==========\n")
  
  # ---------- 8.2 准备预测数据 ----------
  tertile_medians <- df_analysis %>%
    group_by(PLT_group) %>%
    summarise(PLT_RR = median(PLT_RR, na.rm = TRUE)) %>%
    pull(PLT_RR)
  
  newdata_adj <- data.frame(
    PLT_RR = tertile_medians
  )
  
  for (var in final_vars) {
    if (var != "PLT_RR" && var %in% colnames(df_analysis)) {
      if (is.numeric(df_analysis[[var]])) {
        newdata_adj[[var]] <- median(df_analysis[[var]], na.rm = TRUE)
      } else {
        mode_val <- names(sort(table(df_analysis[[var]]), decreasing = TRUE))[1]
        newdata_adj[[var]] <- mode_val
      }
    }
  }
  
  newdata_adj$PLT_group <- c("Low (T1)", "Medium (T2)", "High (T3)")
  
  cat("调整后生存曲线的预测数据（协变量固定为中位数/众数）:\n")
  print(newdata_adj)
  cat("\n")
  
  # ---------- 8.3 计算调整后的生存曲线 ----------
  fit_adj <- survfit(final_cox, newdata = newdata_adj)
  
  # ---------- 8.4 提取调整后的HR ----------
  adj_vars <- setdiff(final_vars, "PLT_RR")
  adj_formula <- paste("Surv(Days_90, Death_90) ~ PLT_group +", paste(adj_vars, collapse = " + "))
  
  df_adj_cox <- df_analysis %>%
    select(all_of(c("Days_90", "Death_90", "PLT_group", adj_vars))) %>%
    mutate(across(where(is.numeric), ~ ifelse(is.na(.), median(., na.rm = TRUE), .)))
  
  cox_adj_tertile <- coxph(as.formula(adj_formula), data = df_adj_cox)
  
  hr_adj <- exp(coef(cox_adj_tertile))
  ci_adj <- exp(confint(cox_adj_tertile))
  p_adj <- summary(cox_adj_tertile)$coefficients[, "Pr(>|z|)"]
  
  hr_adj_tert <- hr_adj[grepl("PLT_group", names(hr_adj))]
  ci_adj_tert <- ci_adj[grepl("PLT_group", rownames(ci_adj)), ]
  p_adj_tert <- p_adj[grepl("PLT_group", names(p_adj))]
  
  hr_adj_annotation <- sprintf(
    "Adjusted HR (Multivariate COX)\nMedium vs Low: HR = %.2f (95%% CI: %.2f-%.2f), p = %.3f\nHigh vs Low: HR = %.2f (95%% CI: %.2f-%.2f), p = %.3f",
    hr_adj_tert[1], ci_adj_tert[1, 1], ci_adj_tert[1, 2], p_adj_tert[1],
    hr_adj_tert[2], ci_adj_tert[2, 1], ci_adj_tert[2, 2], p_adj_tert[2]
  )
  
  # ---------- 8.5 绘制调整后的生存曲线 ----------
  p_adj <- ggsurvplot(
    fit_adj,
    data = newdata_adj,
    conf.int = TRUE,
    conf.int.style = "step",
    legend.labs = c("Low (T1)", "Medium (T2)", "High (T3)"),
    legend.title = "PLT_RR Group",
    xlab = "Survival Time (Days)",
    ylab = "Survival Probability",
    title = "Adjusted Survival Curve by PLT_RR (Tertile Split)",
    subtitle = paste0(
      "Adjusted for: ", 
      paste(setdiff(final_vars, "PLT_RR"), collapse = ", ")
    ),
    palette = c("#2E9FDF", "#F39C12", "#E74C3C"),
    xlim = c(0, 90),
    break.time.by = 10,
    ggtheme = theme_bw() +
      theme(
        legend.position = c(0.82, 0.82),
        legend.background = element_rect(fill = "white", color = "black", size = 0.3),
        legend.title = element_text(size = 10),
        legend.text = element_text(size = 9),
        plot.title = element_text(hjust = 0.5, size = 14, face = "bold"),
        plot.subtitle = element_text(hjust = 0.5, size = 9),
        axis.title = element_text(size = 12),
        axis.text = element_text(size = 10)
      )
  )
  
  p_adj$plot <- p_adj$plot + 
    annotate(
      "text",
      x = 2,
      y = 0.08,
      label = hr_adj_annotation,
      hjust = 0,
      vjust = 0,
      size = 3.0,
      fontface = "plain",
      color = "black"
    )
  
  print(p_adj)
  
  png("04_Adjusted_KM_curve_PLT_RR.png", width = 3000, height = 3600, res = 400)
  print(p_adj)
  dev.off()
  cat("调整后的生存曲线已保存: 04_Adjusted_KM_curve_PLT_RR.png\n\n")
  
  # ---------- 8.6 输出调整后的HR表 ----------
  cat("========== 调整后三分位COX的HR汇总 ==========\n")
  adj_hr_table <- data.frame(
    Comparison = c("Medium (T2) vs Low (T1)", "High (T3) vs Low (T1)"),
    HR = round(hr_adj_tert, 3),
    HR_lower = round(ci_adj_tert[, 1], 3),
    HR_upper = round(ci_adj_tert[, 2], 3),
    P_value = round(p_adj_tert, 4)
  )
  print(adj_hr_table)
  cat("\n")
  
  write.csv(adj_hr_table, "05_Adjusted_tertile_HR_table.csv", row.names = FALSE)
  cat("调整后三分位HR表已保存: 05_Adjusted_tertile_HR_table.csv\n\n")
  
} else {
  cat("\n跳过调整后生存曲线：PLT_RR不在最终模型中\n")
}


# ================================================================
# Part 9: 最终汇总
# ================================================================

cat("============================================================\n")
cat("========== 全部分析完成！ ==========\n")
cat("============================================================\n")
cat("\n生成的文件列表:\n")
cat("  1. 01_Univariate_COX_results.csv - 单因素COX结果\n")
cat("  2. 02_Final_COX_HR_table.csv - 多因素COX的HR汇总表\n")
cat("  3. 03_KM_curve_PLT_RR_unadjusted.png - 未调整的KM生存曲线\n")
cat("  4. 04_Adjusted_KM_curve_PLT_RR.png - 调整后的生存曲线\n")
cat("  5. 05_Adjusted_tertile_HR_table.csv - 调整后三分位HR表\n")

cat("\n========== 主要发现 ==========\n")
if ("PLT_RR" %in% final_vars) {
  plt_rr_hr <- hr_table[hr_table$Variable == "PLT_RR", ]
  cat(sprintf(
    "PLT_RR (连续变量): HR = %.3f (95%% CI: %.3f-%.3f), p = %.4f\n",
    plt_rr_hr$HR, plt_rr_hr$HR_lower, plt_rr_hr$HR_upper, plt_rr_hr$P_value
  ))
}

if (exists("adj_hr_table")) {
  cat("\nPLT_RR三分位（调整后）:\n")
  print(adj_hr_table)
}

cat("\n============================================================\n")
