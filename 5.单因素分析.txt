# 单因素Logistic回归函数
univariate_logistic <- function(data, var, outcome = "Death_binary") {
  # 检查变量是否存在
  if (!var %in% colnames(data)) {
    return(NULL)
  }
  
  # 检查变量是否有足够的变异
  if (length(unique(na.omit(data[[var]]))) < 2) {
    return(NULL)
  }
  
  # 检查是否有足够的样本
  if (sum(!is.na(data[[var]])) < 10) {
    return(NULL)
  }
  
  # 构建公式
  formula <- as.formula(paste(outcome, "~", var))
  
  # 尝试拟合模型
  tryCatch({
    model <- glm(formula, data = data, family = binomial(link = "logit"))
    
    # 提取结果
    summary_model <- summary(model)
    coef_df <- as.data.frame(summary_model$coefficients)
    
    # 提取感兴趣的统计量（排除截距）
    if (nrow(coef_df) > 1) {
      # 对于分类变量，提取所有水平的OR
      results <- data.frame(
        Variable = var,
        Level = rownames(coef_df)[-1],
        Beta = coef_df[-1, "Estimate"],
        SE = coef_df[-1, "Std. Error"],
        Z = coef_df[-1, "z value"],
        P_value = coef_df[-1, "Pr(>|z|)"],
        OR = exp(coef_df[-1, "Estimate"]),
        OR_lower = exp(coef_df[-1, "Estimate"] - 1.96 * coef_df[-1, "Std. Error"]),
        OR_upper = exp(coef_df[-1, "Estimate"] + 1.96 * coef_df[-1, "Std. Error"])
      )
      
      # 对于二分类变量，添加整体P值
      if (nrow(results) == 1) {
        results$Overall_P <- coef_df[-1, "Pr(>|z|)"]
      } else {
        # 使用ANOVA比较模型
        null_model <- glm(as.formula(paste(outcome, "~ 1")), data = data, family = binomial)
        anova_result <- anova(null_model, model, test = "Chisq")
        results$Overall_P <- anova_result[2, "Pr(>Chi)"]
      }
      
      return(results)
    } else {
      return(NULL)
    }
  }, error = function(e) {
    return(NULL)
  })
}

  }

# 创建更详细的单因素分析表格
create_univariate_table <- function(data, vars, outcome = "Death_binary") {
  results_list <- list()
  
  for (var in vars) {
    if (!var %in% colnames(data)) next
    
    # 检查是否为分类变量
    if (is.factor(data[[var]]) || is.character(data[[var]])) {
      data[[var]] <- as.factor(data[[var]])
      levels_var <- levels(data[[var]])
      
      for (i in 2:length(levels_var)) {
        # 创建二分类变量
        temp_data <- data
        temp_data$var_binary <- ifelse(temp_data[[var]] == levels_var[i], 1, 0)
        
        # 计算OR
        tryCatch({
          model <- glm(as.formula(paste(outcome, "~ var_binary")), 
                       data = temp_data, family = binomial)
          coef_sum <- summary(model)$coefficients
          
          if (nrow(coef_sum) > 1) {
            results_list[[paste0(var, "_", levels_var[i])]] <- data.frame(
              Variable = var,
              Level = levels_var[i],
              Reference = levels_var[1],
              N = sum(!is.na(temp_data[[var]])),
              Events = sum(temp_data[[var]] == levels_var[i] & temp_data[[outcome]] == 1, na.rm = TRUE),
              OR = exp(coef_sum[2, "Estimate"]),
              CI_lower = exp(coef_sum[2, "Estimate"] - 1.96 * coef_sum[2, "Std. Error"]),
              CI_upper = exp(coef_sum[2, "Estimate"] + 1.96 * coef_sum[2, "Std. Error"]),
              P_value = coef_sum[2, "Pr(>|z|)"]
            )
          }
        }, error = function(e) {})
      }
    } else if (is.numeric(data[[var]])) {
      # 数值变量
      tryCatch({
        model <- glm(as.formula(paste(outcome, "~", var)), 
                     data = data, family = binomial)
        coef_sum <- summary(model)$coefficients
        
        if (nrow(coef_sum) > 1) {
          results_list[[var]] <- data.frame(
            Variable = var,
            Level = "per 1 unit increase",
            Reference = NA,
            N = sum(!is.na(data[[var]])),
            Events = sum(data[[var]] & data[[outcome]] == 1, na.rm = TRUE),
            OR = exp(coef_sum[2, "Estimate"]),
            CI_lower = exp(coef_sum[2, "Estimate"] - 1.96 * coef_sum[2, "Std. Error"]),
            CI_upper = exp(coef_sum[2, "Estimate"] + 1.96 * coef_sum[2, "Std. Error"]),
            P_value = coef_sum[2, "Pr(>|z|)"]
          )
        }
      }, error = function(e) {})
    }
  }
  
  if (length(results_list) > 0) {
    results_df <- do.call(rbind, results_list)
    results_df$P_adjusted <- p.adjust(results_df$P_value, method = "fdr")
    results_df$Significant <- results_df$P_adjusted < 0.05
    return(results_df)
  } else {
    return(NULL)
  }
}

# 选择P < 0.1的变量进行可视化
if (!is.null(univariate_table)) {
  plot_vars <- univariate_table[univariate_table$P_value < 0.1, ]
  
  if (nrow(plot_vars) > 0) {
    # 准备森林图数据
    forest_data <- plot_vars %>%
      mutate(
        Variable_Level = ifelse(is.na(Level), Variable, paste0(Variable, ": ", Level)),
        OR_display = round(OR, 2),
        CI_display = paste0(round(CI_lower, 2), "-", round(CI_upper, 2)),
        P_display = ifelse(P_value < 0.001, "<0.001", round(P_value, 4))
      )
    
    # 创建森林图
    forest_plot <- ggplot(forest_data, aes(x = OR, y = reorder(Variable_Level, OR))) +
      geom_vline(xintercept = 1, linetype = "dashed", color = "gray50") +
      geom_errorbarh(aes(xmin = CI_lower, xmax = CI_upper), 
                     height = 0.2, color = "steelblue", size = 1) +
      geom_point(aes(color = P_value < 0.05), size = 3) +
      scale_color_manual(values = c("TRUE" = "red", "FALSE" = "steelblue"),
                         labels = c("TRUE" = "P < 0.05", "FALSE" = "P ≥ 0.05")) +
      scale_x_log10() +
      labs(title = "Univariate Analysis for Death (All Variables)",
           x = "Odds Ratio (95% CI)",
           y = "",
           color = "Significance") +
      theme_minimal() +
      theme(
        axis.text.y = element_text(size = 8),
        plot.title = element_text(hjust = 0.5, size = 14, face = "bold"),
        legend.position = "top"
      )
    
    # 保存森林图
    ggsave("univariate_forest_plot_death_all_vars.png", forest_plot, width = 12, height = 10, dpi = 300)
    cat("森林图已保存为 univariate_forest_plot_death_all_vars.png\n")
    
    # 显示森林图
    print(forest_plot)
  }
}
